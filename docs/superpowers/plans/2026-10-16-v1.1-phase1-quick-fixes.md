```ts
/** The Discord name, even after a Google link has overwritten user_metadata. */
export function discordName(identities: UserIdentity[] | undefined): string | null {
  const data = identities?.find((i) => i.provider === "discord")?.identity_data;
  const name = data?.full_name ?? data?.name;
  return typeof name === "string" && name ? name.replace(/#0$/, "") : null;
}
```

- [ ] **Step 4: Run it, expect PASS** — `npx vitest run src/lib/chula.test.ts`.

- [ ] **Step 1: `exact` in NavLinks** (generalizes the existing `"/"` special case):

```tsx
export function NavLinks({ links }: { links: { label: string; href: string; exact?: boolean }[] }) {
  const path = usePathname();
  return links.map((l) => {
    const active = l.exact || l.href === "/" ? path === l.href : path === l.href || path.startsWith(`${l.href}/`);
    return <Link key={l.href} href={l.href} aria-current={active ? "page" : undefined}>{l.label}</Link>;
  });
}
```
