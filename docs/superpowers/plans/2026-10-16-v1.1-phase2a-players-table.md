# v1.1 Phase 2a — Players Table, Filters, Bulk Actions, CSV, Daily Stats

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Admins can scroll every player in one table, filter by status from the dashboard numbers, select rows for bulk actions, export the current list as CSV, and the site starts recording daily player counts for future charts.

**Architecture:** One new security-definer RPC `admin_list_players` (filter + search + sort + offset paging, total via window count) is the single data source for the table, infinite scroll, CSV export and dashboard links. The page is a server component rendering the first page; a small client component appends further pages through a server action and handles row selection. Bulk actions are server actions that loop the existing per-user RPCs (`admin_mark_guest`, `admin_ban_user`) so every existing role rule still applies; "Award" reuses the award page's `sel` URL param. A `stats_daily` table is filled by `pg_cron`.

**Tech Stack:** Next.js 16 App Router (server actions, route handlers — read `node_modules/next/dist/docs/` first), React 19, Supabase Postgres + pgTAP (`npx supabase test db`), Vitest, CSS Modules.

**Spec:** `docs/superpowers/specs/2026-10-16-v1.1-roadmap.md` (Phase 2 + Phase 6 snapshot pulled forward)

**Out of scope (Phase 2b):** Ctrl+K search, customer-support editing on the user page, moderator / event-staff roles.

## Global Constraints

- Every new DB object: RLS on / no table grants; RPCs `security definer set search_path = ''`, `revoke all … from public, anon`, grant `authenticated` only, first line re-checks `current_app_role() in ('owner','admin')` → `FORBIDDEN`.
- Filter values exactly: `all | chula_verified | guest | unverified | staff | banned`. Player-status filters count `role = 'user'` only, so they match `admin_overview_stats` numbers (dashboard "81 Verified Chula" → table shows the same 81).
- Page size 50; CSV page size 1000; bulk cap 200 ids per action.
- CSV cells starting with `= + - @` (or tab/CR) are prefixed with `'` (spreadsheet formula injection).
- Loading / empty / error states on the table. Infinite scroll also works without JS via a "Load more" link.
- No new dependencies. No commits without kaikub's approval; no Claude attribution lines.
- After code changes: `npm run db:types`, `graphify update .`.

## Review Focus

1. Dashboard number and filtered table count disagree (e.g. staff counted in "Verified Chula").
2. Search text with `%`, `_` or `\` must match literally, not as wildcards.
3. A bulk action where some ids fail (admin targeting an admin, already banned) must still apply the rest and report "N done, M skipped".
4. Infinite scroll must not duplicate or skip rows when two pages load quickly (stable sort tie-breaker on `id`).
5. CSV export must honour the same filter + search as the screen and must not be reachable by non-admins.

```sql
-- One list behind the admin players table, its infinite scroll, the CSV export
-- and the dashboard number links. Status filters count role = 'user' only, the
-- same rule as admin_overview_stats, so "81 Verified Chula" lists those 81.
--
-- ponytail: offset paging. Fine for a few thousand players; switch to a
-- (created_at, id) keyset if the table ever feels slow deep in the scroll.
create or replace function public.admin_list_players(
  p_filter text default 'all', p_query text default '', p_sort text default 'joined',
  p_desc boolean default true, p_offset int default 0, p_limit int default 50)
returns table (
  user_id uuid, display_name text, handle text, role text, email text, chula_email text,
  minecraft_usernames text, verification_kind text, banned boolean, created_at timestamptz, total bigint)
language plpgsql stable security definer set search_path = '' as $$
#variable_conflict use_column
declare
  v_query text := coalesce(trim(p_query), '');
  v_pattern text := '%' || replace(replace(replace(v_query, '\', '\\'), '%', '\%'), '_', '\_') || '%';
begin
  if coalesce(public.current_app_role(), '') not in ('owner', 'admin') then raise exception 'FORBIDDEN'; end if;
  if p_filter not in ('all', 'chula_verified', 'guest', 'unverified', 'staff', 'banned')
     or p_sort not in ('joined', 'name') then raise exception 'INVALID'; end if;

  return query
  with base as (
    select u.id, coalesce(p.role, 'user') as role, u.email::text as email, c.email as chula_email,
      public.admin_display_name(u.id) as display_name,
      coalesce(d.name, u.email::text) as handle,
      case when public.is_chula_verified(u.id) then 'verified'
           when p.guest_verified_at is not null then 'guest'
           else 'unverified' end as kind,
      public.is_banned(u.id) as banned,
      u.created_at
    from auth.users u
    left join public.profiles p on p.user_id = u.id
    left join public.chula_claims c on c.user_id = u.id
    left join lateral (
      select coalesce(i.identity_data ->> 'full_name', i.identity_data ->> 'user_name', i.identity_data ->> 'name') as name
      from auth.identities i where i.user_id = u.id and i.provider = 'discord' limit 1
    ) d on true
    where v_query = ''
      or u.email ilike v_pattern or c.email ilike v_pattern
      or exists (
        select 1 from auth.identities i
        where i.user_id = u.id and i.provider = 'discord'
          and (i.provider_id = v_query
            or i.identity_data ->> 'full_name' ilike v_pattern
            or i.identity_data ->> 'user_name' ilike v_pattern
            or i.identity_data ->> 'name' ilike v_pattern
            or i.identity_data -> 'custom_claims' ->> 'global_name' ilike v_pattern))
      or exists (select 1 from auth.identities g where g.user_id = u.id and g.provider = 'google' and g.identity_data ->> 'email' ilike v_pattern)
      or exists (select 1 from public.minecraft_registrations r where r.user_id = u.id and r.minecraft_username ilike v_pattern)
  )
  select b.id, b.display_name, b.handle, b.role, b.email, b.chula_email,
    (select string_agg(r.minecraft_username, ', ' order by r.created_at)
       from public.minecraft_registrations r where r.user_id = b.id and r.is_active),
    b.kind, b.banned, b.created_at, count(*) over ()
  from base b
  where case p_filter
    when 'all' then true
    when 'staff' then b.role <> 'user'
    when 'banned' then b.banned
    when 'chula_verified' then b.role = 'user' and b.kind = 'verified'
    when 'guest' then b.role = 'user' and b.kind = 'guest'
    when 'unverified' then b.role = 'user' and b.kind = 'unverified'
  end
  order by
    case when p_sort = 'name' and not p_desc then lower(b.display_name) end asc,
    case when p_sort = 'name' and p_desc then lower(b.display_name) end desc,
    case when p_sort = 'joined' and not p_desc then b.created_at end asc,
    case when p_sort = 'joined' and p_desc then b.created_at end desc,
    b.id
  offset greatest(coalesce(p_offset, 0), 0)
  limit least(greatest(coalesce(p_limit, 50), 1), 1000);
end;
$$;
revoke all on function public.admin_list_players(text, text, text, boolean, int, int) from public, anon;
grant execute on function public.admin_list_players(text, text, text, boolean, int, int) to authenticated;
```

`src/lib/csv.ts`:

```ts
type Cell = string | number | boolean | null;

/** RFC 4180 CSV. Text starting like a formula gets a leading ' so a
 *  spreadsheet shows it instead of running it. */
export function toCsv(rows: Cell[][]): string {
  const cell = (value: Cell) => {
    if (value === null) return "";
    let text = String(value);
    if (typeof value === "string" && /^[=+\-@\t\r]/.test(text)) text = `'${text}`;
    return /[",\r\n]/.test(text) ? `"${text.replace(/"/g, '""')}"` : text;
  };
  return rows.map((row) => row.map(cell).join(",") + "\r\n").join("");
}
```

```ts
export const FILTERS = {
  all: "Everyone",
  chula_verified: "Verified Chula",
  guest: "Guests",
  unverified: "Unverified",
  staff: "Staff",
  banned: "Banned",
} as const;

export type Filter = keyof typeof FILTERS;
export type PlayerSearch = { filter?: string; q?: string; sort?: string; dir?: string };

/** URL search params → admin_list_players args. Unknown values fall back to the
 *  defaults, so a hand-edited URL never reaches the database as an INVALID call. */
export function playerListArgs(sp: PlayerSearch) {
  const p_filter: Filter = sp.filter && sp.filter in FILTERS ? (sp.filter as Filter) : "all";
  const p_sort: "joined" | "name" = sp.sort === "name" ? "name" : "joined";
  return { p_filter, p_query: sp.q?.trim() ?? "", p_sort, p_desc: sp.dir !== "asc" };
}
```

```ts
import { toCsv } from "@/lib/csv";
import { createClient } from "@/lib/supabase/server";
import { playerListArgs } from "../params";

const PAGE = 1000;

export async function GET(request: Request) {
  const sp = Object.fromEntries(new URL(request.url).searchParams);
  const args = playerListArgs(sp);
  const supabase = await createClient();
  const rows: (string | boolean | null)[][] = [["user_id", "name", "discord", "role", "status", "banned", "email", "chula_email", "minecraft", "joined"]];
  for (let offset = 0; ; offset += PAGE) {
    const { data, error } = await supabase.rpc("admin_list_players", { ...args, p_offset: offset, p_limit: PAGE });
    if (error) return new Response(error.message.includes("FORBIDDEN") ? "Forbidden" : "Export failed", { status: error.message.includes("FORBIDDEN") ? 403 : 500 });
    for (const r of data) rows.push([r.user_id, r.display_name, r.handle, r.role, r.verification_kind, r.banned, r.email, r.chula_email, r.minecraft_usernames, r.created_at]);
    if (data.length < PAGE) break;
  }
  const day = new Date().toISOString().slice(0, 10);
  return new Response(toCsv(rows), {
    headers: {
      "Content-Type": "text/csv; charset=utf-8",
      "Content-Disposition": `attachment; filename="chulacraft-players-${args.p_filter}-${day}.csv"`,
      "Cache-Control": "no-store",
    },
  });
}
```

```ts
"use server";

import { createClient } from "@/lib/supabase/server";
import type { Database } from "@/lib/supabase/database.types";
import { playerListArgs, type PlayerSearch } from "./params";

export type PlayerRow = Database["public"]["Functions"]["admin_list_players"]["Returns"][number];
export const PAGE_SIZE = 50;

export async function loadPlayers(search: PlayerSearch, offset: number) {
  const supabase = await createClient();
  const { data, error } = await supabase.rpc("admin_list_players", {
    ...playerListArgs(search), p_offset: Math.max(0, Math.floor(offset) || 0), p_limit: PAGE_SIZE,
  });
  if (error) return { error: true as const };
  return { rows: data, total: Number(data[0]?.total ?? 0) };
}
```

```tsx
  const [rows, setRows] = useState(initial);
```

```ts
import { revalidatePath } from "next/cache";
import { dbErrorCode } from "@/lib/db-error";
import { UUID } from "@/lib/registration";

const MAX_BULK = 200;
export type BulkResult = { done: number; skipped: { id: string; code: string }[] } | { error: "INVALID" | "TOO_MANY" };

// One RPC per player on purpose: each call re-applies that RPC's role rules
// (admins can't touch admins, no self-ban), so bulk can't do more than single.
async function each(ids: string[], call: (id: string) => PromiseLike<{ error: { message: string } | null }>): Promise<BulkResult> {
  if (ids.length > MAX_BULK) return { error: "TOO_MANY" };
  if (ids.length === 0 || !ids.every((id) => UUID.test(id))) return { error: "INVALID" };
  const skipped: { id: string; code: string }[] = [];
  for (const id of new Set(ids)) {
    const { error } = await call(id);
    if (error) skipped.push({ id, code: dbErrorCode(error) ?? "FAILED" });
  }
  revalidatePath("/admin/players");
  return { done: new Set(ids).size - skipped.length, skipped };
}

export async function bulkMarkGuest(ids: string[]) {
  const supabase = await createClient();
  return each(ids, (id) => supabase.rpc("admin_mark_guest", { p_user_id: id }));
}

export async function bulkBan(ids: string[], reason: string) {
  const text = reason.trim();
  if (!text) return { error: "INVALID" as const };
  const supabase = await createClient();
  return each(ids, (id) => supabase.rpc("admin_ban_user", { p_user_id: id, p_reason: text, p_public_note: null, p_expires_at: null }));
}
```

```sql
-- One row per day of headline counts, so charts can show history that the
-- live tables can't (players removed or banned later still count on the day).
-- History starts the day this ships.
create table if not exists public.stats_daily (
  day date primary key,
  players int not null, verified int not null, guests int not null, unverified int not null,
  whitelisted int not null, banned int not null,
  created_at timestamptz not null default now()
);
alter table public.stats_daily enable row level security;
revoke all on public.stats_daily from anon, authenticated, service_role;

create or replace function public.snapshot_stats_daily()
returns void language plpgsql security definer set search_path = '' as $$
begin
  insert into public.stats_daily (day, players, verified, guests, unverified, whitelisted, banned)
  select (now() at time zone 'Asia/Bangkok')::date,
    count(*) filter (where coalesce(p.role, 'user') = 'user'),
    count(*) filter (where coalesce(p.role, 'user') = 'user' and public.is_chula_verified(u.id)),
    count(*) filter (where coalesce(p.role, 'user') = 'user' and not public.is_chula_verified(u.id) and p.guest_verified_at is not null),
    count(*) filter (where coalesce(p.role, 'user') = 'user' and not public.is_chula_verified(u.id) and p.guest_verified_at is null),
    (select count(*) from public.minecraft_registrations r where r.is_active and r.desired_whitelisted),
    count(*) filter (where public.is_banned(u.id))
  from auth.users u left join public.profiles p on p.user_id = u.id
  on conflict (day) do update set
    players = excluded.players, verified = excluded.verified, guests = excluded.guests,
    unverified = excluded.unverified, whitelisted = excluded.whitelisted, banned = excluded.banned, created_at = now();
end;
$$;
revoke all on function public.snapshot_stats_daily() from public, anon, authenticated;

create or replace function public.admin_stats_daily(p_days int default 90)
returns setof public.stats_daily language plpgsql stable security definer set search_path = '' as $$
begin
  if coalesce(public.current_app_role(), '') not in ('owner', 'admin') then raise exception 'FORBIDDEN'; end if;
  return query select * from public.stats_daily s
    where s.day > current_date - least(greatest(coalesce(p_days, 90), 1), 730)
    order by s.day;
end;
$$;
revoke all on function public.admin_stats_daily(int) from public, anon;
grant execute on function public.admin_stats_daily(int) to authenticated;

-- 23:55 Bangkok = 16:55 UTC.
create extension if not exists pg_cron;
select cron.schedule('stats-daily', '55 16 * * *', 'select public.snapshot_stats_daily()');
select public.snapshot_stats_daily();
```
